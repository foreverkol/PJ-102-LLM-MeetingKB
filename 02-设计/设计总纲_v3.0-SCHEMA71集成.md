### 3.1 模块 1:`entity_resolver.py`(v7.0 强制新增)

```python
"""
v7.0 FR-v7.0-003 实体统一编号 + FR-v7.0-004 canonical_name + aliases

entity_id 格式:person_{canonical_name_hash8}_{seq4}
registry: SYSTEM/registry/entity_registry.json(append-only)
"""

import hashlib
import json
from pathlib import Path


class EntityResolver:
    def __init__(self, registry_path: Path):
        self.registry_path = registry_path
        self.registry = self._load()

    def resolve_or_create(
        self,
        entity_type: str,      # "person" | "organization"
        raw_name: str,
        aliases: list = None,
        context: dict = None
    ) -> dict:
        # 1. 规范化
        canonical = self._normalize(raw_name)
        # 2. 查 registry
        existing = self._find(canonical, aliases or [])
        if existing:
            return {
                "entity_id": existing["entity_id"],
                "canonical_name": existing["canonical_name"],
                "aliases": list(set(existing["aliases"] + (aliases or []))),
                "action": "update",
                "disambiguation_candidates": [],
            }
        # 3. 新建
        pinyin_prefix = self._to_pinyin_prefix(canonical)
        seq = self._next_seq(entity_type, pinyin_prefix)
        entity_id = f"{entity_type}_{pinyin_prefix}_{seq:04d}"
        return {
            "entity_id": entity_id,
            "canonical_name": canonical,
            "aliases": aliases or [],
            "action": "create",
            "disambiguation_candidates": self._find_similar(canonical),
        }

    def _normalize(self, name): return name.strip().replace(" ", "")
    def _to_pinyin_prefix(self, name):
        # 简化:用 hashlib.md5 前 8 字符(生产环境应用 pypinyin)
        return hashlib.md5(name.encode()).hexdigest()[:8]
    def _find(self, canonical, aliases):
        for e in self.registry.get("entities", []):
            if e["canonical_name"] == canonical:
                return e
            if any(a in e.get("aliases", []) for a in aliases):
                return e
        return None
    def _next_seq(self, entity_type, prefix):
        existing = [int(e["entity_id"].split("_")[-1])
                    for e in self.registry.get("entities", [])
                    if e["entity_id"].startswith(f"{entity_type}_{prefix}")]
        return max(existing, default=0) + 1
    def _find_similar(self, canonical):
        return [{"canonical_name": e["canonical_name"], "match_score": 0.5}
                for e in self.registry.get("entities", [])
                if canonical in e["canonical_name"] or e["canonical_name"] in canonical
                ][:3]
    def _load(self):
        if self.registry_path.exists():
            return json.loads(self.registry_path.read_text())
        return {"entities": []}
    def save(self):
        self.registry_path.write_text(
            json.dumps(self.registry, ensure_ascii=False, indent=2))
```

### 3.2 模块 2:`lifecycle_stage.py`(v7.0 状态机)

```python
"""
v7.0 FR-v7.0-005 status_stage 5 阶段状态机

5 阶段合法跃迁:
  raw       → compiled
  compiled  → reviewed | superseded
  reviewed  → canonical | superseded
  canonical → superseded
  superseded → (终结态)
"""

from enum import Enum
from datetime import datetime


class Stage(Enum):
    RAW = "raw"
    COMPILED = "compiled"
    REVIEWED = "reviewed"
    CANONICAL = "canonical"
    SUPERSEDED = "superseded"


TRANSITIONS = {
    Stage.RAW: {Stage.COMPILED},
    Stage.COMPILED: {Stage.REVIEWED, Stage.SUPERSEDED},
    Stage.REVIEWED: {Stage.CANONICAL, Stage.SUPERSEDED},
    Stage.CANONICAL: {Stage.SUPERSEDED},
    Stage.SUPERSEDED: set(),
}


def transition_validation(current, target):
    return target in TRANSITIONS.get(current, set())


def update_with_version(frontmatter: dict, source_ref: str) -> dict:
    """v7.0 §8.4 版本规则"""
    current_version = frontmatter.get("version", 1)
    new_version = current_version + 1
    log_entry = {
        "version": current_version,
        "superseded_at": datetime.now().isoformat(),
        "by_source": source_ref,
    }
    frontmatter["version"] = new_version
    frontmatter["updated"] = datetime.now().date().isoformat()
    frontmatter.setdefault("previous_versions", []).append(log_entry)
    return frontmatter
```

### 3.3 模块 3:`scenario_extractor.py`(v7.0 新页面)

```python
"""
v7.0 FR-v7.0-008 scenario 11 字段提取
"""

SCENARIO_PROMPT = """分析以下会议内容,识别是否存在"商业模式片段"(scenario):
- theme: 业务主题/模式名
- customer: 目标客户
- pain_point: 客户痛点
- offering: 提供的产品/服务
- value_capture: 价值捕获方式
- channel: 分销渠道
- key_resources: 关键资源
- key_constraints: 关键约束
- hidden_assumptions: 隐性假设
- trigger_signals: 触发信号
- failure_modes: 失败模式

输出 JSON:
  - 若识别到至少 3 个清晰字段,返回完整 scenario
  - 若识别不到,返回 null
"""


def extract_scenarios(content: str, llm) -> list:
    """从单条会议内容提取 0-N 个 scenario"""
    response = llm.call(SCENARIO_PROMPT + content, max_tokens=2500)
    data = json.loads(response)
    return data.get("scenarios", []) if data else []
```
